STEP 5: Test Correct Solution & Confetti

Goal: Prove that a valid algorithm passes all tests and triggers rewards.

1. Paste a working Bubble Sort solution:

function bubbleSort(arr: number[]): number[] {
  const n = arr.length;
  for (let i = 0; i < n - 1; i++) {
    for (let j = 0; j < n - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        const temp = arr[j];
        arr[j] = arr[j + 1];
        arr[j + 1] = temp;
      }
    }
  }
  return arr;
}
